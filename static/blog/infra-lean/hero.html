infra: infrastructure as code in Lean 4

One apply, four resources, no depends_on: a required reference and two addresses the cloud has not assigned yet, and the order falls out of both.

fleet cluster in paris where
  resource aws securityGroup "web" as web
    { description := "http and https, ssh from nowhere" }
  resource aws awsInstance "app-1" as app1
    { imageId := "latest", instanceType := InstanceType.of .t3 .nano
    , securityGroup := web }
  resource aws awsInstance "app-2" as app2
    { imageId := "latest", instanceType := InstanceType.of .t3 .nano
    , securityGroup := web }
  resource aws secrets "cluster-peers"
    { valueFrom := composed expr!"{privateIpOf app1},{privateIpOf app2}" }
